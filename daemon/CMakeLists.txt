# The whole framework has one version, in VERSION at the repository root. A source tree exported
# from daemon/ alone (the Debian source package) cannot see the root, so it carries a copy beside
# this file instead; the root file wins when both exist.
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../VERSION")
  set(HOLDER_VERSION_FILE "${CMAKE_CURRENT_SOURCE_DIR}/../VERSION")
else()
  set(HOLDER_VERSION_FILE "${CMAKE_CURRENT_SOURCE_DIR}/VERSION")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${HOLDER_VERSION_FILE}")
file(READ "${HOLDER_VERSION_FILE}" HOLDER_DAEMON_VERSION)
string(STRIP "${HOLDER_DAEMON_VERSION}" HOLDER_DAEMON_VERSION)
if(HOLDER_DAEMON_VERSION MATCHES "^([0-9]+)\\.([0-9]+)\\.([0-9]+)(-[0-9A-Za-z-]+(\\.[0-9A-Za-z-]+)*)?(\\+[0-9A-Za-z-]+(\\.[0-9A-Za-z-]+)*)?$")
  # CMake's project(VERSION) accepts only numeric components. Keep the full
  # SemVer for the daemon's reported version and expose its numeric core to
  # CMake's project version machinery.
  set(HOLDER_DAEMON_CMAKE_VERSION
      "${CMAKE_MATCH_1}.${CMAKE_MATCH_2}.${CMAKE_MATCH_3}")
else()
  message(FATAL_ERROR
    "${HOLDER_VERSION_FILE} must contain a semantic version like 0.1.0 or 2.0.0-dev")
endif()

# api/ (the HTTP contract: openapi.yaml and API_VERSION), cli/ (holderctl) and common/ (the platform
# files holderd and holderctl share) live beside daemon/ in the framework repository. A Debian
# source package is exported from daemon/ alone, so its workflow copies the three directories in;
# the ones inside this directory are used when the ones next to it are out of reach.
foreach(dir api cli common)
  string(TOUPPER "${dir}" upper)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../${dir}")
    set(HOLDER_${upper}_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../${dir}")
  elseif(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${dir}")
    set(HOLDER_${upper}_DIR "${CMAKE_CURRENT_SOURCE_DIR}/${dir}")
  else()
    message(FATAL_ERROR
      "The ${dir}/ directory was not found next to or inside ${CMAKE_CURRENT_SOURCE_DIR}")
  endif()
endforeach()

function(holder_stage_windows_runtime target)
  if(WIN32 AND HOLDER_CORE_SDK AND NOT HOLDER_CORE_SOURCE_DIR AND NOT HOLDER_USE_SYSTEM_CORE)
    file(GLOB runtime_dlls CONFIGURE_DEPENDS
      "${HOLDER_CORE_SDK}/bin/*.dll"
      "${VCPKG_INSTALLED_DIR}/x64-windows/bin/*.dll")
    if(runtime_dlls)
      add_custom_command(TARGET ${target} POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different
          ${runtime_dlls} "$<TARGET_FILE_DIR:${target}>"
        COMMAND_EXPAND_LISTS)
    endif()
  endif()
endfunction()

# GCC 12 and 13 report -Warray-bounds inside fmt v9's basic_memory_buffer growth code (reached
# through spdlog) when optimising; it is a false positive in the library, and GCC 14 does not
# report it. Only those compilers are told to skip the check, so newer GCC and Clang keep it.
set(HOLDER_GCC_FMT_FALSE_POSITIVE "")
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 14)
  set(HOLDER_GCC_FMT_FALSE_POSITIVE -Wno-array-bounds)
endif()

# Warnings (GNU/Clang)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(holderd PRIVATE
    -Wall -Wextra -Wpedantic
    -Wshadow -Wconversion -Wsign-conversion
    -Wno-unknown-pragmas
    ${HOLDER_GCC_FMT_FALSE_POSITIVE}
  )
  if(HOLDER_WARNINGS_AS_ERRORS)
    target_compile_options(holderd PRIVATE -Werror)
  endif()
endif()

# GCC 13 reports a false -Wstringop-overflow in fmt 9's bigint buffer growth.
# Limit the workaround to the affected source and dependency/compiler versions.
# https://github.com/fmtlib/fmt/issues/3533
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU"
   AND CMAKE_CXX_COMPILER_VERSION VERSION_GREATER_EQUAL 13
   AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 14
   AND fmt_VERSION VERSION_GREATER_EQUAL 9
   AND fmt_VERSION VERSION_LESS 10)
  set(HOLDER_FMT_WARNING_TARGETS holderd)
  if(TARGET holder_daemon_tests)
    list(APPEND HOLDER_FMT_WARNING_TARGETS holder_daemon_tests)
  endif()
  set_property(SOURCE src/api/routes/ai/runs/AiRunPostRoute.cpp
    TARGET_DIRECTORY ${HOLDER_FMT_WARNING_TARGETS}
    APPEND PROPERTY COMPILE_OPTIONS -Wno-stringop-overflow)
endif()
